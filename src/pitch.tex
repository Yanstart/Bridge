% --- 8. Couche transport & securite (ch.4 + ch.7) --------------------
\chref{Ch.4 \& Ch.7 --- Transport \& s\'ecurit\'e}
\begin{frame}{Transport s\'ecuris\'e}
  \begin{itemize}
    \item mTLS sur la sortie HAPI FHIR. Authentification mutuelle.
    \item File de repli SQLite chiffr\'ee si r\'eseau coup\'e. RTO \textless\ 30 s.
    \item Logs structur\'es JSON, sans donn\'ee patient. Minimisation par construction.
  \end{itemize}
  \note{Slide 8. La securite repose sur 3 piliers. mTLS pour le transport. Une file de repli locale chiffree pour la resilience. Et surtout, aucune donnee clinique dans nos logs. Le dashboard regarde les flux, pas le contenu.}
\end{frame}
