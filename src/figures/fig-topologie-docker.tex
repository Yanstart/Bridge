% =====================================================================
% figures/fig-topologie-docker.tex
% Topologie Docker Compose : 4 simulateurs + bridge + HAPI + MQTT broker.
% Layout aere : etiquettes de protocole entre la rangee des sims et le bus,
% bridge centre, dashboard et HAPI sur la meme rangee, auteur exterieur.
% =====================================================================
\begin{tikzpicture}[
  font=\sffamily,
  every node/.style={font=\small\sffamily},
  >={Stealth[length=2.5mm]},
  thick,
  sim/.style={
    rectangle, rounded corners=3pt,
    draw=bridgealert, fill=bridgealert!10, text=bridgealert,
    minimum width=2.6cm, minimum height=1.0cm,
    align=center, font=\small\sffamily
  },
  bus/.style={
    rectangle, rounded corners=3pt,
    draw=bridgeteal, fill=bridgeteal!12, text=bridgeteal,
    minimum width=2.6cm, minimum height=0.9cm,
    align=center, font=\small\sffamily
  },
  vol/.style={
    rectangle, rounded corners=3pt,
    draw=bridgegray, dashed, fill=bridgegray!8,
    minimum width=2.6cm, minimum height=0.9cm,
    align=center, font=\small\sffamily, text=bridgegray
  },
  bridgecore/.style={
    rectangle, rounded corners=4pt,
    draw=bridgeblue, fill=bridgeblue!12, text=bridgeblue,
    minimum width=6.6cm, minimum height=1.6cm,
    align=center, font=\bfseries\sffamily
  },
  dashbox/.style={
    rectangle, rounded corners=3pt,
    draw=bridgeteal, fill=bridgeteal!15, text=bridgeteal,
    minimum width=2.6cm, minimum height=1.6cm,
    align=center, font=\small\bfseries\sffamily
  },
  hapi/.style={
    rectangle, rounded corners=3pt,
    draw=bridgegreen, fill=bridgegreen!15, text=bridgegreen,
    minimum width=3.0cm, minimum height=1.0cm,
    align=center, font=\bfseries\sffamily
  },
  protolbl/.style={
    rectangle, rounded corners=2pt,
    fill=white, draw=bridgegray!40,
    font=\scriptsize\itshape\sffamily, color=bridgegray,
    inner sep=2pt
  },
  flowarrow/.style={-{Stealth[length=3mm]}, thick, draw=bridgeblue!80}
]

% =====================================================================
% Cadre du reseau Docker (un peu plus grand pour aerer)
% =====================================================================
\node[draw=bridgeblue, dashed, thick, fill=bridgesoft,
      rounded corners=4pt,
      minimum width=16.4cm, minimum height=11.0cm,
      anchor=south west,
      label={[anchor=north west, font=\footnotesize\bfseries,
              color=bridgeblue, xshift=8pt, yshift=-6pt]
              north west: Reseau Docker bridge-demo (\texttt{docker compose up})}]
      (net) at (0, 0) {};

% =====================================================================
% Niveau 1 : 4 simulateurs alignes en haut
% =====================================================================
\def\Ysim{9.4}
\node[sim] (sim1) at (2.5, \Ysim) {sim-monitor\\Philips};
\node[sim] (sim2) at (5.8, \Ysim) {sim-glucometer\\Masimo};
\node[sim] (sim3) at (9.4, \Ysim) {sim-spirometer\\Welch Allyn};
\node[sim] (sim4) at (13.4, \Ysim) {sim-pump\\Drager};

% =====================================================================
% Niveau 1.5 : protocoles, bien separes des sims, fond blanc
% =====================================================================
\def\Yproto{8.05}
\node[protolbl] at (2.5, \Yproto) {TCP HL7v2};
\node[protolbl] at (5.8, \Yproto) {MQTT JSON};
\node[protolbl] at (9.4, \Yproto) {fichier XML};
\node[protolbl] at (13.4, \Yproto) {RS-232 (TCP simule)};

% =====================================================================
% Niveau 2 : Bus MQTT et volume partage (alignes sous leur sim)
% =====================================================================
\def\Ybus{6.6}
\node[bus] (mqtt) at (5.8, \Ybus) {MQTT broker\\eclipse-mosquitto};
\node[vol] (vol)  at (9.4, \Ybus) {Volume \texttt{drop/}};

% =====================================================================
% Niveau 3 : Bridge core (centre-gauche) et Dashboard (droite)
% =====================================================================
\def\Ybridge{4.2}
\node[bridgecore] (bridge) at (5.8, \Ybridge) {IoT Edge Bridge\\Python + FastAPI};
\node[font=\scriptsize\itshape, color=bridgegray, anchor=north]
  at (5.8, 3.55) {Adapter . Parser . Mapper . Transport};

\node[dashbox] (dashb) at (12.4, \Ybridge) {Dashboard\\:8080};

% =====================================================================
% Niveau 4 : HAPI FHIR Server (en bas, decale a gauche)
% =====================================================================
\def\Yhapi{1.6}
\node[hapi] (hapi) at (5.8, \Yhapi) {HAPI FHIR Server\\:8081};

% =====================================================================
% Acteur exterieur (auteur), juste au-dessus du dashboard
% =====================================================================
\node[anchor=south, font=\footnotesize\itshape, color=bridgegray]
  (user) at (12.4, 6.0) {Auteur (navigateur)};

% =====================================================================
% Fleches : simulateurs vers leurs cibles
% =====================================================================
% sim1 (TCP HL7v2) directement vers Bridge (ancrage gauche)
\draw[flowarrow] (sim1.south) -- ++(0, -0.5)
                 -| ([xshift=-2.2cm]bridge.north);

% sim2 (MQTT) -> broker MQTT (verticale propre)
\draw[flowarrow] (sim2.south) -- (mqtt.north);

% sim3 (fichier) -> volume drop
\draw[flowarrow] (sim3.south) -- (vol.north);

% sim4 (RS-232) directement vers Bridge (ancrage droit)
\draw[flowarrow] (sim4.south) -- ++(0, -0.5)
                 -| ([xshift=2.2cm]bridge.north);

% Broker MQTT et volume vers Bridge
\draw[flowarrow] (mqtt.south)  -- ([xshift=-0.7cm]bridge.north);
\draw[flowarrow] (vol.south)   -- ([xshift=0.7cm]bridge.north);

% Bridge -> HAPI (verticale, label a droite)
\draw[flowarrow] (bridge.south) -- (hapi.north)
  node[midway, right=2pt, font=\scriptsize\itshape, color=bridgegray] {POST FHIR};

% Bridge -> Dashboard (horizontale, label au-dessus)
\draw[flowarrow] (bridge.east) -- (dashb.west)
  node[midway, above, font=\scriptsize\itshape, color=bridgegray] {WebSocket};

% Auteur -> Dashboard (en pointille, hors reseau Docker conceptuellement)
\draw[->, dashed, thick, color=bridgegray]
  (user.south) -- (dashb.north)
  node[midway, right=2pt, font=\scriptsize\itshape, color=bridgegray] {HTTP};

\end{tikzpicture}
