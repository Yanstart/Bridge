  task/.style={

  bridgetask/.style={

  errtask/.style={
    task,
    draw=bridgealert, fill=bridgealert!10, text=bridgealert

    rectangle, draw=bridgegray, dashed, fill=bridgegray!5,

  % --- Flux ---------------------------------------------------------
  seqflow/.style={-{Stealth[length=2.5mm]}, thick, draw=bridgeblue!80},
  errflow/.style={-{Stealth[length=2.5mm]}, thick, dashed, draw=bridgealert},
  msgflow/.style={-{Stealth[length=2.5mm]}, thick, dotted, draw=bridgegray!90},

% Cadre du pool

% Etiquettes des lanes (zone gauche du pool)

  \node[anchor=west, font=\tiny\itshape\sffamily, color=bridgegray]

}
