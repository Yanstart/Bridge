\section{1. Le probleme}

\chref{Ch.1 --- Le probl\`eme}
\begin{frame}{Le probl\`eme : 0~\% au stade 4}
  \begin{itemize}
    \item Plan eSant\'e belge 2025-2027 cible le BIHR comme colonne vert\'ebrale.
    \item Aucun h\^opital \'echantillonn\'e n'int\`egre automatiquement ses dispositifs au DPI.
    \item Le parc Legacy a 15 \`a 20 ans, les standards changent tous les 3 \`a 5 ans.
  \end{itemize}
  \vspace{2mm}
  \begin{bridgewarn}[Verrou MDR 2017/745]
    Modifier un dispositif certifi\'e supprime son marquage CE. On ne peut donc rien toucher au mat\'eriel.
  \end{bridgewarn}
  \note{Le BIHR est en marche, mais le terrain ne suit pas. Personne n'integre automatiquement. Et on ne peut pas remplacer le parc, car le MDR interdit toute modification d'un dispositif certifie. Voila notre verrou de depart.}
\end{frame}

\chref{Ch.1 --- Processus m\'etier}
\begin{frame}{Le processus m\'etier qu'on automatise}
  \begin{center}
    \resizebox{\linewidth}{!}{\input{figures/fig-processus-metier.tex}}\\[2mm]
    {\itshape\small 4 acteurs, 6 \'etapes, aujourd'hui 100~\% manuel.}
  \end{center}
  \note{Aujourd'hui, transferer une mesure du moniteur au DPI passe par un soignant qui retape. C'est lent, c'est risque, ca ne passe pas a l'echelle. Notre passerelle remplace ces six etapes manuelles par un flux continu et trace.}
\end{frame}

\section{2. Applications et fonctionnalit\'es}

\chref{Ch.4 --- Notre r\'eponse}
\begin{frame}{Notre r\'eponse : un middleware configurable}
  \begin{center}
    \resizebox{0.88\textwidth}{!}{\input{figures/fig-pipeline-4couches.tex}}\\[2mm]
    {\itshape\small Quatre couches enfichables. Une question simple par dispositif : quel canal, quelle langue, quelle sortie ?}
  \end{center}
  \note{Quatre couches. Adapter capte le canal. Parser decode la langue. Mapper codifie en LOINC et FHIR. Transport publie de maniere securisee. Pour ajouter un dispositif, on repond a trois questions dans un fichier JSON. Aucune ligne de code.}
\end{frame}

\chref{Ch.2 \& Ch.7 --- Dashboard OT}
\begin{frame}{Le dashboard, vu c\^ot\'e technicien biom\'edical}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \fbox{\parbox{0.95\linewidth}{\centering\itshape\small
        Capture du dashboard : grille de 5 cartes \'equipement (vert/orange/rouge),
        compteurs OT (\'equipements connect\'es, messages trait\'es, quarantaine),
        timeline activit\'e, l\'egende permanente.}}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{itemize}
        \item Statut couleur en 1 coup d'\oe il
        \item Compteurs temps r\'eel
        \item \'Edition des profils JSON
        \item Aucune valeur clinique patient
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1mm}
  \begin{bridgekey}[R\`egle de minimisation]
    Le dashboard regarde les flux, pas le contenu clinique. Les techniciens biom\'edicaux voient \emph{que} \c{c}a circule, pas \emph{ce qui} circule.
  \end{bridgekey}
  \note{Le dashboard parle aux techniciens biomedicaux et a l'OT, pas aux soignants. Il montre les flux, jamais la valeur patient. C'est un choix structurant : la donnee clinique passe au DPI, le dashboard garde l'oeil sur le tuyau.}
\end{frame}

\section{6. \'Evaluation et perspectives}

\chref{Ch.6 --- \'Evaluation}
\begin{frame}{\'Evaluation : o\`u en sommes-nous ?}
  \begin{itemize}
    \item Statut juridique : middleware d'int\'egration IT, \textbf{non-DM}.
    \item Pyramide mhealth.belgium : niveau M2 vis\'e en d\'emo (s\'ecurit\'e + interop\'erabilit\'e).
    \item Argument m\'edico-\'economique : facteur 5 moins cher qu'un parc neuf.
  \end{itemize}
  \vspace{1mm}
  \begin{center}
    \scriptsize
    \begin{tabular}{ll}
      \toprule
      \textbf{KPI} & \textbf{Cible} \\
      \midrule
      Latence p95 ingestion                                       & moins de 500 ms \\
      Taux de transmission r\'eussie                              & sup\'erieur \`a 99~\% \\
      Minutes \'economis\'ees par soignant et par jour            & sup\'erieur \`a 30 \\
      Lignes de code modifi\'ees pour ajouter un dispositif       & 0 \\
      \bottomrule
    \end{tabular}
  \end{center}
  \note{Statut non-DM, ca simplifie tout. Niveau M2 vise. Cinq fois moins cher qu'un remplacement de parc. Et surtout, zero ligne de code pour ajouter un nouveau dispositif. C'est mesurable, c'est tenable, c'est notre engagement.}
\end{frame}

\chref{Roadmap}
\begin{frame}{Trois \'evolutions sur la roadmap}
  \begin{itemize}
    \item F1. Plugins de traduction communautaires (mod\`ele HACS, npm).
    \item F2. Anomaly detection en bord (IA l\'eg\`ere ONNX).
    \item F3. F\'ed\'eration europ\'eenne EHDS (mobilit\'e transfrontali\`ere).
  \end{itemize}
  \vspace{2mm}
  \begin{bridgeinfo}[Effet de catalogue]
    Aucun \'editeur ne peut couvrir seul tous les dispositifs m\'edicaux du monde. La communaut\'e \'ecrit ses plugins. Le bridge devient un standard de fait.
  \end{bridgeinfo}
  \note{La passerelle est concue pour grandir. Plugins communautaires comme HACS pour Home Assistant. IA legere en bord pour pre-flagger les aberrations. Federation EHDS pour la mobilite europeenne. Le code est pret a recevoir tout ca.}
\end{frame}

\section{7. Securite, confidentialite, ethique}

\chref{Ch.7 --- S\'ecurit\'e}
\begin{frame}{S\'ecurit\'e et confidentialit\'e par construction}
  \begin{center}
    \begin{tabular}{ll}
      \toprule
      \textbf{Pilier} & \textbf{Mise en \oe uvre} \\
      \midrule
      Confidentialit\'e & TLS 1.3, principe de minimisation, dashboard sans donn\'ee patient \\
      Int\'egrit\'e     & Checksum SHA-256 sur file de repli, validation FHIR avant \'emission \\
      Disponibilit\'e   & Retry exponentiel, fallback SQLite, hot-reload des profils \\
      Tra\c{c}abilit\'e & Logs JSON horodat\'es UTC, sans valeur clinique \\
      \bottomrule
    \end{tabular}
  \end{center}
  \note{Securite sur les quatre piliers CIA plus tracabilite. TLS 1.3 partout. Checksum sur tout ce qui transite. Retry et fallback pour la dispo. Logs sans valeur clinique pour respecter la minimisation. Rien n'est laisse au hasard.}
\end{frame}

\chref{Ch.7 --- \'Ethique}
\begin{frame}{Trois questions \'ethiques}
  \begin{itemize}
    \item Responsabilit\'e du mapping : qui est responsable si une unit\'e est mal cod\'ee ?
    \item Consentement au partage hub : distinct du consentement RGPD g\'en\'eral.
    \item Risque de datafication : capter facilement n'invite pas \`a capter sans limite.
  \end{itemize}
  \vspace{2mm}
  \begin{bridgewarn}[AIPD obligatoire en prod]
    Avant tout d\'eploiement r\'eel, une analyse d'impact RGPD formelle, avec consultation du DPO de l'\'etablissement, est obligatoire (RGPD art. 35).
  \end{bridgewarn}
  \note{Trois questions qu'on ne peut pas eluder. Qui est responsable si une unite est traduite a l'envers. Le consentement hub regional est distinct du RGPD general. Et le risque de tout capter parce qu'on en a la capacite. La technique ne dispense pas de la reflexion.}
\end{frame}
